\section*{Dimensionless numbers}
$\mathrm{Re}=UL/\nu$, $\Pr=\nu/\alpha$, $\mathrm{Sc}_k=\nu/D_k$, $\mathrm{Le}_k=\alpha/D_k$,
$\mathrm{Da}=\tau_{\rm flow}/\tau_{\rm chem}$, $\beta=E(T_b-T_u)/\Ru T_b^2$ (Zeldovich), $\mathrm{Ma}=u/c$.

\section*{Further reading}
\begin{itemize}
  \item S.~R.~Turns, \emph{An Introduction to Combustion}, McGraw-Hill: the gentlest complete textbook.
  \item T.~Poinsot \& D.~Veynante, \emph{Theoretical and Numerical Combustion}: the governing equations,
        flame speeds and flamelets, for the CFD-minded.
  \item N.~Peters, \emph{Turbulent Combustion}, CUP: mixture fraction and flamelets.
  \item C.~K.~Law, \emph{Combustion Physics}, CUP: stretch, Lewis-number effects, hydrogen.
  \item ANSYS Chemkin Theory Manual: exact forms of every property and rate expression.
  \item Cantera documentation and examples: \url{https://cantera.org}.
  \item ASEC onboarding: \url{https://asec-laboratory.github.io/Onboarding/04_ReactiveFlows_and_Cantera/00_Introduction/}.
\end{itemize}
